/* Example 4.6 Combining blocking and non-blocking tasks.
 * Two continuous tasks at priority 1, one periodic task (3 ms) at priority 2.
 * Note: a continuous task prints in a tight loop, so the output is large. */

static const char * pcTextForTask1 = "Continuous task 1 running";
static const char * pcTextForTask2 = "Continuous task 2 running";

void vContinuousProcessingTask( void * pvParameters )
{
    char * pcTaskName = ( char * ) pvParameters;

    for( ;; )
    {
        vPrintLine( pcTaskName );
    }
}

void vPeriodicTask( void * pvParameters )
{
    TickType_t xLastWakeTime;
    const TickType_t xDelay3ms = pdMS_TO_TICKS( 3 );

    xLastWakeTime = xTaskGetTickCount();

    for( ;; )
    {
        vPrintLine( "Periodic task is running" );
        vTaskDelayUntil( &xLastWakeTime, xDelay3ms );
    }
}

int main( void )
{
    xTaskCreate( vContinuousProcessingTask, "Task 1", 1000, ( void * ) pcTextForTask1, 1, NULL );
    xTaskCreate( vContinuousProcessingTask, "Task 2", 1000, ( void * ) pcTextForTask2, 1, NULL );
    xTaskCreate( vPeriodicTask, "Task 3", 1000, NULL, 2, NULL );

    vTaskStartScheduler();

    for( ;; );
}
